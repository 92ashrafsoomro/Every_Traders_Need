<template>
    <!-- TITLE BAR -->
    <user-title-bar>
        <h1 class="text-h3 mb-2 font-weight-bold">Smart Auction Search</h1>
        <p class="text-subtitle-1 mb-2 font-weight-medium">
            Filter, compare, and uncover vehicles that match your profit goals.
        </p>

        <div class="d-flex mt-6 ga-3">
            <v-btn variant="tonal" class="text-none px-5" :class="{ 'bg-primary text-white': auctionStore.auctionTab }"
                @click="auctionStore.toggleAuctionTab()">
                Auction Finder
            </v-btn>

            <v-btn variant="tonal" class="text-none px-5" :class="{ 'bg-primary text-white': !auctionStore.auctionTab }"
                @click="auctionStore.toggleAuctionTab()">
                Vehicle Valuation
            </v-btn>
        </div>
    </user-title-bar>


    <v-navigation-drawer v-model="auctionStore.sidebar" location="start" temporary elevation="10" :width="drawerWidth"
        style="top: 70px;">
        <auctionSidebar />
    </v-navigation-drawer>

    <v-container fluid class="pt-0 pb-0" >
        <v-container style="max-width:1400px">
            <v-row class="mt-2 ">
                <v-col cols="12">

                    <!-- ROW 1 -->
                    <div class="d-flex flex-wrap align-center ga-3">

                        <!-- LEFT -->
                        <div class="d-flex align-center ga-3 flex-grow-1">
                            <v-btn color="primary" variant="outlined" prepend-icon="mdi-filter"
                                @click="auctionStore.toggleFilter()">
                                {{ auctionStore.sidebar ? 'Hide Filter' : 'Show Filter' }}
                            </v-btn>


                            <v-slide-x-transition>
                                <v-btn v-if="auctionStore.sidebar" color="danger" variant="text"
                                    prepend-icon="mdi-delete" :disabled="!isAnyFilterSelected"
                                    @click="auctionStore.ClearFilter()">
                                    Clear All
                                </v-btn>
                            </v-slide-x-transition>

                            <div>
                                {{ auctionStore.offset }} -
                                {{ auctionStore.offset + auctionStore.filter.length }}
                                of {{ auctionStore.total }} Vehicles
                            </div>
                        </div>

                        <!-- RIGHT -->
                        <div class="d-flex flex-wrap align-center ga-3 ml-auto">
                            <v-select v-model="auctionStore.filter.length" density="compact" variant="outlined"
                                label="Length" :items="[10, 50, 100, 200, 500]" style="width:120px"
                                @update:model-value="handleInput" />

                            <v-select v-model="auctionStore.filter.sort_by" density="compact" variant="outlined"
                                label="Sort by" item-title="name" item-value="id" :items="sortingOptions"
                                style="width:260px" @update:model-value="handleInput" />
                        </div>
                    </div>


                    <v-slide-y-transition>
                        <div v-if="!auctionStore.sidebar && isAnyFilterSelected" class=" mt-3">
                            <div class="d-flex flex-wrap align-center ga-3 ">

                                <!-- CHIPS -->
                                <div class="d-flex align-center overflow-x-auto">
                                    <Chips />
                                </div>


                                <v-btn color="danger" variant="text" prepend-icon="mdi-delete"
                                    @click="auctionStore.ClearFilter()">
                                    Clear All
                                </v-btn>

                            </div>
                        </div>
                    </v-slide-y-transition>

                </v-col>
            </v-row>
        </v-container>
    </v-container>


    <v-container fluid class="mt-n6">
        <v-container style="max-width:1400px">
            <component :is="currentComponent" />
        </v-container>
    </v-container>
</template>


<script>
import AuctionFinder from "./auctionDetail.vue";
import VehicleValuation from "./vehicleValuation.vue";
import auctionSidebar from "./sidebar/index.vue";
import Chips from "./Chips.vue";
import { useAuctionStore } from "@/stores/auctionStore";
import { useDisplay } from 'vuetify'

export default {
    components: {
        AuctionFinder,
        VehicleValuation,
        auctionSidebar,
        Chips
    },

    data() {
        return {
            auctionStore: useAuctionStore(),
            display : useDisplay()
        };
    },

    mounted() {
        this.auctionStore.loadSiderBarFilters();
        this.auctionStore.getAuctionList();
    },

    computed: {
        currentComponent() {
            return this.auctionStore.auctionTab
                ? "AuctionFinder"
                : "VehicleValuation";
        },
 drawerWidth() {
   
      return this.display.smAndDown ? '100%' : 330
    },
        sortingOptions() {
            return [
                { id: "name-asc", name: "Name: A to Z" },
                { id: "name-dsc", name: "Name: Z to A" },
                { id: "grade-desc", name: "Grade: Low to High" },
                { id: "grade-asc", name: "Grade: High to Low" },
                { id: "date-desc", name: "Date & Time: Oldest First" },
                { id: "date-asc", name: "Date & Time: Newest First" }
            ];
        },

        isAnyFilterSelected() {
            const filter = this.auctionStore.filter;
            const ignoreKeys = ["length", "offset", "sort_by", "page"];

            return Object.keys(filter).some(key => {
                if (ignoreKeys.includes(key)) return false;
                const value = filter[key];
                return Array.isArray(value)
                    ? value.length > 0
                    : value !== null && value !== "";
            });
        }
    },

    methods: {
        handleInput() {
            this.auctionStore.getAuctionList();
        }
    }
};
</script>
